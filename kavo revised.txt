local ThemeSubscribers = {}

local function SubscribeToTheme(Instance: Instance, Property: string, ColorKey: string)
	if not Instance or not Instance.Parent then return end
	table.insert(ThemeSubscribers, {
		Instance = Instance,
		Property = Property,
		ColorKey = ColorKey
	})
	local Color = ActiveTheme[ColorKey]
	if Color then
		Instance[Property] = Color
	end
end

local function UpdateThemeColors(NewTheme: {[string]: Color3})
	for Key, Color in pairs(NewTheme) do
		ActiveTheme[Key] = Color
	end
	for I = #ThemeSubscribers, 1, -1 do
		local Sub = ThemeSubscribers[I]
		local Inst = Sub.Instance
		if not Inst or not Inst.Parent then
			table.remove(ThemeSubscribers, I)
		else
			local NewColor = ActiveTheme[Sub.ColorKey]
			if NewColor and Inst[Sub.Property] ~= NewColor then
				Inst[Sub.Property] = NewColor
			end
		end
	end
end

local Utility = {}

function Utility:TweenObject(Obj: Instance, Properties: {[string]: any}, Duration: number, ...)
	TweenService:Create(Obj, TweenInfo.new(Duration, ...), Properties):Play()
end

function Utility:RippleEffect(Button: GuiButton, SampleTemplate: ImageLabel)
	local C = SampleTemplate:Clone()
	C.Parent = Button
	local X = Mouse.X - C.AbsolutePosition.X
	local Y = Mouse.Y - C.AbsolutePosition.Y
	C.Position = UDim2.new(0, X, 0, Y)
	local MaxDim = math.max(Button.AbsoluteSize.X, Button.AbsoluteSize.Y) * 1.5
	local TargetSize = UDim2.new(0, MaxDim, 0, MaxDim)
	local TargetPos = UDim2.new(0.5, -MaxDim / 2, 0.5, -MaxDim / 2)
	C:TweenSizeAndPosition(TargetSize, TargetPos, Enum.EasingDirection.Out, Enum.EasingStyle.Quad, 0.35, true)
	TweenService:Create(C, TweenInfoRipple, {ImageTransparency = 1}):Play()
	task.delay(0.35, function()
		C:Destroy()
	end)
end

function Kavo:DraggingEnabled(Frame: Frame, Parent: Frame?)
	local DragParent = Parent or Frame
	local Dragging = false
	local DragInput: InputObject? = nil
	local MousePos: Vector2 = Vector2.zero
	local FramePos: UDim2 = UDim2.new()

	local function OnInputChanged(Input: InputObject)
		if Input == DragInput and Dragging then
			local Delta = Input.Position - MousePos
			DragParent.Position = UDim2.new(
				FramePos.X.Scale, FramePos.X.Offset + Delta.X,
				FramePos.Y.Scale, FramePos.Y.Offset + Delta.Y
			)
		end
	end

	local InputChangedConnection = UserInputService.InputChanged:Connect(OnInputChanged)

	Frame.InputBegan:Connect(function(Input: InputObject)
		if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then
			Dragging = true
			MousePos = Input.Position
			FramePos = DragParent.Position
			DragInput = Input
			Input.Changed:Connect(function()
				if Input.UserInputState == Enum.UserInputState.End then
					Dragging = false
					DragInput = nil
				end
			end)
		end
	end)

	Frame.InputChanged:Connect(function(Input: InputObject)
		if Input.UserInputType == Enum.UserInputType.MouseMovement or Input.UserInputType == Enum.UserInputType.Touch then
			DragInput = Input
		end
	end)

	DragParent.AncestryChanged:Connect(function(_, NewParent)
		if not NewParent then
			InputChangedConnection:Disconnect()
		end
	end)
end

				local function ZigZag(X: number): number
					return math.acos(math.cos(X * math.pi)) / math.pi
				end

				local function UpdateColorPicker(InputPosition: Vector2)
					if ColorPickerActive then
						local X = math.clamp(InputPosition.X - Rgb.AbsolutePosition.X, 0, Rgb.AbsoluteSize.X)
						local Y = math.clamp(InputPosition.Y - Rgb.AbsolutePosition.Y, 0, Rgb.AbsoluteSize.Y)
						local NX = X / Rgb.AbsoluteSize.X
						local NY = Y / Rgb.AbsoluteSize.Y
						RgbCircle.Position = UDim2.fromScale(NX, NY)
						CurrentColor = {1 - NX, 1 - NY, CurrentColor[3]}
						local RealColor = Color3.fromHSV(CurrentColor[1], CurrentColor[2], CurrentColor[3])
						ColorCurrent.BackgroundColor3 = RealColor
						pcall(Callback, RealColor)
					end
					if DarknessActive then
						local Y = math.clamp(InputPosition.Y - Darkness.AbsolutePosition.Y, 0, Darkness.AbsoluteSize.Y)
						local NY = Y / Darkness.AbsoluteSize.Y
						DarkCircle.Position = UDim2.fromScale(0.5, NY)
						DarkCircle.ImageColor3 = Color3.fromHSV(0, 0, NY)
						CurrentColor = {CurrentColor[1], CurrentColor[2], 1 - NY}
						local RealColor = Color3.fromHSV(CurrentColor[1], CurrentColor[2], CurrentColor[3])
						ColorCurrent.BackgroundColor3 = RealColor
						pcall(Callback, RealColor)
					end
				end

				local function SetColor(Tbl: {number})
					CurrentColor = {Tbl[1], Tbl[2], Tbl[3]}
					RgbCircle.Position = UDim2.fromScale(1 - CurrentColor[1], 1 - CurrentColor[2])
					DarkCircle.Position = UDim2.fromScale(0.5, 1 - CurrentColor[3])
					DarkCircle.ImageColor3 = Color3.fromHSV(0, 0, 1 - CurrentColor[3])
					local RealColor = Color3.fromHSV(CurrentColor[1], CurrentColor[2], CurrentColor[3])
					ColorCurrent.BackgroundColor3 = RealColor
				end

				local function SetRgbColor(Tbl: {number})
					CurrentColor = {Tbl[1], Tbl[2], CurrentColor[3]}
					RgbCircle.Position = UDim2.fromScale(1 - CurrentColor[1], 1 - CurrentColor[2])
					local RealColor = Color3.fromHSV(CurrentColor[1], CurrentColor[2], CurrentColor[3])
					ColorCurrent.BackgroundColor3 = RealColor
					pcall(Callback, RealColor)
				end

				local function ToggleRainbow()
					if Rainbow then
						TweenService:Create(ToggleEnabled, TweenInfo.new(0.1, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut), {ImageTransparency = 1}):Play()
						Rainbow = false
						if RainbowConnection then RainbowConnection:Disconnect() end
					else
						TweenService:Create(ToggleEnabled, TweenInfo.new(0.1, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut), {ImageTransparency = 0}):Play()
						Rainbow = true
						RainbowConnection = RunService.RenderStepped:Connect(function()
							SetRgbColor({ZigZag(Counter), 1, 1})
							Counter += 0.01
						end)
					end
				end
