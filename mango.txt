local function ClickEffect(Object)
	local Clone = Sample:Clone()
	Clone.Parent = Object
	Clone.Position = UDim2.new(0, Mouse.X - Clone.AbsolutePosition.X, 0, Mouse.Y - Clone.AbsolutePosition.Y)
	
	local Size = math.max(Object.AbsoluteSize.X, Object.AbsoluteSize.Y) * 1.5
	Clone:TweenSizeAndPosition(
		UDim2.new(0, Size, 0, Size), 
		UDim2.new(0.7, -Size / 2, 0.5, -Size / 2), 
		Enum.EasingDirection.Out, Enum.EasingStyle.Quad, 0.35, true
	)
	
	for _ = 1, 10 do
		Clone.ImageTransparency += 0.05
		task.wait(0.35 / 12)
	end
	Clone:Destroy()
end

local function MakeDraggable(TopbarObject, TargetObject)
	local Dragging, DragInput, DragStart, StartPosition
	
	TopbarObject.InputBegan:Connect(function(Input)
		if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then
			Dragging = true
			DragStart = Input.Position
			StartPosition = TargetObject.Position
			
			Input.Changed:Connect(function()
				if Input.UserInputState == Enum.UserInputState.End then Dragging = false end
			end)
		end
	end)
	
	TopbarObject.InputChanged:Connect(function(Input)
		if Input.UserInputType == Enum.UserInputType.MouseMovement or Input.UserInputType == Enum.UserInputType.Touch then
			DragInput = Input
		end
	end)
	
	UserInputService.InputChanged:Connect(function(Input)
		if Input == DragInput and Dragging then
			local Delta = Input.Position - DragStart
			TweenService:Create(TargetObject, TweenInfo.new(0.2), {
				Position = UDim2.new(StartPosition.X.Scale, StartPosition.X.Offset + Delta.X, StartPosition.Y.Scale, StartPosition.Y.Offset + Delta.Y)
			}):Play()
		end
	end)
end
