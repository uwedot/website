local function EnsureFolder(Path: string)
	if not UseStudio and isfolder and not isfolder(Path) then
		pcall(makefolder, Path)
	end
end

local function SafeWrite(Path: string, Content: string)
	if UseStudio then return end
	if writefile then pcall(writefile, Path, Content) end
end

local function SafeRead(Path: string): string?
	if UseStudio then return nil end
	if isfile and isfile(Path) then
		local Ok, Res = pcall(readfile, Path)
		return if Ok then Res else nil
	end
	return nil
end

local CustomAssets = {}
local Icons = nil

local function ResolveIcon(Icon: (number | string)?): (string, Vector2?, Vector2?)
	if not Icon or Icon == 0 then return "", nil, nil end
	
	if typeof(Icon) == "string" and (Icon:find("rbxasset://") == 1 or Icon:find("rbxthumb://") == 1) then
		return Icon, nil, nil
	end
	
	if typeof(Icon) == "string" and Icons then
		local SizedIcons = Icons['48px']
		local R = SizedIcons[string.match(string.lower(Icon), "^%s*(.*)%s*$")]
		if R and type(R[1]) == "number" then
			return "rbxassetid://" .. R[1], Vector2.new(R[3][1], R[3][2]), Vector2.new(R[2][1], R[2][2])
		end
	elseif type(Icon) == "number" then
		return "rbxassetid://" .. Icon, nil, nil
	end
	
	return "", nil, nil
end

local function ChangeTheme(Theme: (string | {[string]: Color3}))
	if typeof(Theme) == 'string' then
		SelectedTheme = RayfieldLibrary.Theme[Theme] or RayfieldLibrary.Theme.Default
	elseif typeof(Theme) == 'table' then
		SelectedTheme = Theme
	else
		SelectedTheme = RayfieldLibrary.Theme.Default
	end
	
	task.spawn(function()
		if Rayfield and Rayfield.Main then
			Rayfield.Main.BackgroundColor3 = SelectedTheme.Background 
		end
	end)
end

local SettingsTable = {
	General = {RayfieldOpen = {Type = 'bind', Value = 'K', Name = 'Rayfield Keybind'}},
	System = {UsageAnalytics = {Type = 'toggle', Value = false, Name = 'Anonymised Analytics'}}
}

local OverriddenSettings = {}

local function GetSetting(Category: string, Name: string): any
	local Key = Category .. "." .. Name
	if OverriddenSettings[Key] ~= nil then return OverriddenSettings[Key] end
	local S = SettingsTable[Category] and SettingsTable[Category][Name]
	return if S then S.Value else nil
end

local function LoadSettings()
	local File = SafeRead(RayfieldFolder..'/settings'..ConfigurationExtension)
	if File then
		local Ok, Decoded = pcall(HttpService.JSONDecode, HttpService, File)
		if Ok and Decoded then
			for Cat, Vals in Decoded do
				if SettingsTable[Cat] then
					for K, V in Vals do
						if SettingsTable[Cat][K] and typeof(V.Value) == typeof(SettingsTable[Cat][K].Value) then
							SettingsTable[Cat][K].Value = V.Value
						end
					end
				end
			end
		end
	end
end

local function SaveSettings()
	if UseStudio then return end
	local Ok, Encoded = pcall(HttpService.JSONEncode, HttpService, SettingsTable)
	if Ok then SafeWrite(RayfieldFolder..'/settings'..ConfigurationExtension, Encoded) end
end

local function MakeDraggable(Object: GuiObject, DragObject: GuiObject)
	local Dragging, Relative, Offset = false, nil, Vector2.zero
	local ScreenGui = Object:FindFirstAncestorWhichIsA("ScreenGui")
	if ScreenGui and ScreenGui.IgnoreGuiInset then
		Offset += game:GetService('GuiService'):GetGuiInset()
	end
	
	DragObject.InputBegan:Connect(function(Input, Processed)
		if Processed then return end
		if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then
			Dragging = true
			Relative = Object.AbsolutePosition + Object.AbsoluteSize * Object.AnchorPoint - UserInputService:GetMouseLocation()
		end
	end)
	
	UserInputService.InputEnded:Connect(function(Input)
		if Dragging and (Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch) then
			Dragging = false
		end
	end)
	
	RunService.RenderStepped:Connect(function()
		if Dragging and not Hidden then
			local Pos = UserInputService:GetMouseLocation() + Relative + Offset
			Object.Position = UDim2.fromOffset(Pos.X, Pos.Y)
			if DragBar then DragBar.Position = UDim2.fromOffset(Pos.X, Pos.Y + DragOffset) end
		end
	end)
end

function RayfieldLibrary:Notify(Data: {Title: string?, Content: string?, Image: (number|string)?, Duration: number?})
	task.spawn(function()
		local Notif = Notifications.Template:Clone()
		Notif.Name = Data.Title or "Notification"
		Notif.Parent = Notifications
		Notif.LayoutOrder = #Notifications:GetChildren()
		Notif.Visible = false
		
		Notif.Title.Text = Data.Title or "Unknown"
		Notif.Description.Text = Data.Content or ""
		
		local Img, RectOff, RectSz = ResolveIcon(Data.Image)
		Notif.Icon.Image = Img
		if RectOff then Notif.Icon.ImageRectOffset = RectOff end
		if RectSz then Notif.Icon.ImageRectSize = RectSz end
		
		Notif.BackgroundColor3 = SelectedTheme.NotificationBackground
		Notif.Title.TextColor3 = SelectedTheme.TextColor
		Notif.Description.TextColor3 = SelectedTheme.TextColor
		Notif.UIStroke.Color = SelectedTheme.TextColor
		Notif.Icon.ImageColor3 = SelectedTheme.TextColor
		
		Notif.Size = UDim2.new(1, 0, 0, 800)
		Notif.Visible = true
		
		local BoundsY = math.max(Notif.Title.TextBounds.Y + Notif.Description.TextBounds.Y + 31, 60)
		TweenService:Create(Notif, TweenInfo.new(0.6, Enum.EasingStyle.Exponential), {Size = UDim2.new(1, 0, 0, BoundsY)}):Play()
		task.wait(0.15)
		TweenService:Create(Notif, TweenInfo.new(0.4, Enum.EasingStyle.Exponential), {BackgroundTransparency = 0.45}):Play()
		TweenService:Create(Notif.Title, TweenInfo.new(0.3, Enum.EasingStyle.Exponential), {TextTransparency = 0}):Play()

local function SetElementsVisible(Show: boolean)
	for _, Tab in ipairs(Elements:GetChildren()) do
		if Tab:IsA("ScrollingFrame") and Tab.Name ~= "Template" and Tab.Name ~= "Placeholder" then
			for _, El in ipairs(Tab:GetChildren()) do
				if El:IsA("Frame") and El.Name ~= "SectionSpacing" and El.Name ~= "Placeholder" then
					if El.Name == "SectionTitle" or El.Name:find("SearchTitle") then
						TweenService:Create(El.Title, TweenInfo.new(0.3, Enum.EasingStyle.Exponential), {TextTransparency = Show and 0.4 or 1}):Play()
					elseif El.Name == "Divider" then
						TweenService:Create(El.Divider, TweenInfo.new(0.3, Enum.EasingStyle.Exponential), {BackgroundTransparency = Show and 0.85 or 1}):Play()
					else
						local BgT = El:GetAttribute("BackgroundTransparencyTarget") or 0
						local StT = El:GetAttribute("UIStrokeTransparencyTarget") or 0

local function SaveConfiguration(FlagOverride: string?)
	if not CEnabled or not GlobalLoaded then return end
	local Data = {}
	for K, V in pairs(RayfieldLibrary.Flags) do
		if V.Type == "ColorPicker" then
			Data[K] = {R = V.Color.R * 255, G = V.Color.G * 255, B = V.Color.B * 255}
		else
			Data[K] = V.CurrentValue or V.CurrentKeybind or V.CurrentOption or V.Color
		end
	end
	SafeWrite(ConfigurationFolder .. "/" .. CFileName .. ConfigurationExtension, HttpService:JSONEncode(Data))
end

local function LoadConfiguration(ConfigStr: string?): boolean
	local Json = ConfigStr or SafeRead(ConfigurationFolder .. "/" .. CFileName .. ConfigurationExtension)
	if not Json then return false end
	
	local Ok, Data = pcall(HttpService.JSONDecode, HttpService, Json)
	if not Ok then return false end
	
	local Changed = false
	for FlagName, Flag in pairs(RayfieldLibrary.Flags) do
		local Val = Data[FlagName]
		if Val ~= nil then
			task.spawn(function()
				if Flag.Type == "ColorPicker" and typeof(Val) == "table" then
					Flag:Set(Color3.fromRGB(Val.R, Val.G, Val.B))
					Changed = true
				elseif Flag.CurrentValue ~= Val or Flag.CurrentKeybind ~= Val or Flag.CurrentOption ~= Val then
					Flag:Set(Val)
					Changed = true
				end
			end)
		end
	end
	return Changed
end

		function Tab:CreateSlider(SliderSettings: {Name: string, Range: {number}, Increment: number, CurrentValue: number?, Suffix: string?, Flag: string?, Callback: (number) -> (), Ext: boolean?})
			local Slider = Elements.Template.Slider:Clone()
			Slider.Name = SliderSettings.Name
			Slider.Title.Text = SliderSettings.Name
			Slider.Parent = TabPage
			Slider.Visible = true
			SliderSettings.CurrentValue = math.clamp(SliderSettings.CurrentValue or SliderSettings.Range[1], SliderSettings.Range[1], SliderSettings.Range[2])
			
			TweenService:Create(Slider, TweenInfo.new(0.7, Enum.EasingStyle.Exponential), {BackgroundTransparency = 0}):Play()
			TweenService:Create(Slider.UIStroke, TweenInfo.new(0.7, Enum.EasingStyle.Exponential), {Transparency = 0}):Play()
			TweenService:Create(Slider.Title, TweenInfo.new(0.7, Enum.EasingStyle.Exponential), {TextTransparency = 0}):Play()
			
			local function UpdateVal(Val: number)
				Val = math.clamp(Val, SliderSettings.Range[1], SliderSettings.Range[2])
				local Pct = (Val - SliderSettings.Range[1]) / (SliderSettings.Range[2] - SliderSettings.Range[1])
				local Width = math.max(Pct * Slider.Main.AbsoluteSize.X, 5)
				TweenService:Create(Slider.Main.Progress, TweenInfo.new(0.45, Enum.EasingStyle.Exponential, Enum.EasingDirection.Out), {Size = UDim2.new(0, Width, 1, 0)}):Play()
				Slider.Main.Information.Text = tostring(Val) .. (SliderSettings.Suffix and (" " .. SliderSettings.Suffix) or "")
			end
			
			UpdateVal(SliderSettings.CurrentValue)
			
			local Dragging = false
			Slider.Main.Interact.InputBegan:Connect(function(Inp)
				if Inp.UserInputType == Enum.UserInputType.MouseButton1 or Inp.UserInputType == Enum.UserInputType.Touch then Dragging = true end
			end)
			Slider.Main.Interact.InputEnded:Connect(function(Inp)
				if Inp.UserInputType == Enum.UserInputType.MouseButton1 or Inp.UserInputType == Enum.UserInputType.Touch then Dragging = false end
			end)
			
			Slider.Main.Interact.MouseButton1Down:Connect(function(X)
				local Start = X
				local Conn
				Conn = RunService.Stepped:Connect(function()
					if not Dragging then Conn:Disconnect(); return end
					local Loc = UserInputService:GetMouseLocation().X
					Loc = math.clamp(Loc, Slider.Main.AbsolutePosition.X, Slider.Main.AbsolutePosition.X + Slider.Main.AbsoluteSize.X)
					local Raw = SliderSettings.Range[1] + ((Loc - Slider.Main.AbsolutePosition.X) / Slider.Main.AbsoluteSize.X) * (SliderSettings.Range[2] - SliderSettings.Range[1])
					local NewVal = math.floor(Raw / SliderSettings.Increment + 0.5) * SliderSettings.Increment
					NewVal = math.clamp(NewVal, SliderSettings.Range[1], SliderSettings.Range[2])
					
					if SliderSettings.CurrentValue ~= NewVal then
						SliderSettings.CurrentValue = NewVal
						UpdateVal(NewVal)
						pcall(SliderSettings.Callback, NewVal)
						if not SliderSettings.Ext then SaveConfiguration() end
					end
				end)
			end)
			
			local Obj = {
				Set = function(Self, Val: number)
					Val = math.clamp(Val, SliderSettings.Range[1], SliderSettings.Range[2])
					SliderSettings.CurrentValue = Val
					UpdateVal(Val)
					pcall(SliderSettings.Callback, Val)
					if not SliderSettings.Ext then SaveConfiguration() end
				end
			}
			
			if SliderSettings.Flag then
				SliderSettings.Type = "Slider"
				RayfieldLibrary.Flags[SliderSettings.Flag] = SliderSettings
			end
			return Obj
		end
		
		function Tab:CreateDropdown(DropdownSettings: {Name: string, Options: {string}, CurrentOption: (string|{string})?, MultipleOptions: boolean?, Flag: string?, Callback: ({string}) -> (), Ext: boolean?})
			local Dropdown = Elements.Template.Dropdown:Clone()
			Dropdown.Name = DropdownSettings.Name
			Dropdown.Title.Text = DropdownSettings.Name
			Dropdown.Parent = TabPage
			Dropdown.Visible = true
			
			if typeof(DropdownSettings.CurrentOption) == "string" then DropdownSettings.CurrentOption = {DropdownSettings.CurrentOption} end
			DropdownSettings.CurrentOption = DropdownSettings.CurrentOption or {}
			if not DropdownSettings.MultipleOptions and #DropdownSettings.CurrentOption > 1 then DropdownSettings.CurrentOption = {DropdownSettings.CurrentOption[1]} end
			
			local function UpdateDisplay()
				if #DropdownSettings.CurrentOption == 0 then Dropdown.Selected.Text = "None"
				elseif #DropdownSettings.CurrentOption == 1 then Dropdown.Selected.Text = DropdownSettings.CurrentOption[1]
				else Dropdown.Selected.Text = "Various" end
				
				for _, Opt in ipairs(Dropdown.List:GetChildren()) do
					if Opt:IsA("Frame") and Opt.Name ~= "Placeholder" and Opt.Name ~= "Template" then
						local Sel = table.find(DropdownSettings.CurrentOption, Opt.Name) ~= nil
						Opt.BackgroundColor3 = Sel and SelectedTheme.DropdownSelected or SelectedTheme.DropdownUnselected
					end
				end
			end
			
			local function Populate()
				for _, Child in ipairs(Dropdown.List:GetChildren()) do
					if Child:IsA("Frame") and Child.Name ~= "Placeholder" and Child.Name ~= "Template" then Child:Destroy() end
				end
				for _, OptName in ipairs(DropdownSettings.Options) do
					local Opt = Elements.Template.Dropdown.List.Template:Clone()
					Opt.Name = OptName
					Opt.Title.Text = OptName
					Opt.Parent = Dropdown.List
					Opt.Visible = true
					Opt.Interact.ZIndex = 50
					
					Opt.Interact.MouseButton1Click:Connect(function()
						local Idx = table.find(DropdownSettings.CurrentOption, OptName)
						if Idx then
							table.remove(DropdownSettings.CurrentOption, Idx)
						else
							if not DropdownSettings.MultipleOptions then table.clear(DropdownSettings.CurrentOption) end
							table.insert(DropdownSettings.CurrentOption, OptName)
						end
						UpdateDisplay()
						pcall(DropdownSettings.Callback, DropdownSettings.CurrentOption)
						if not DropdownSettings.Ext then SaveConfiguration() end
						
						if not DropdownSettings.MultipleOptions then
							task.wait(0.1)
							TweenService:Create(Dropdown, TweenInfo.new(0.5, Enum.EasingStyle.Exponential), {Size = UDim2.new(1, -10, 0, 45)}):Play()
							Dropdown.List.Visible = false
							TweenService:Create(Dropdown.Toggle, TweenInfo.new(0.7, Enum.EasingStyle.Exponential), {Rotation = 180}):Play()
						end
					end)
				end
				UpdateDisplay()
			end
			
			Populate()
			
			TweenService:Create(Dropdown, TweenInfo.new(0.7, Enum.EasingStyle.Exponential), {BackgroundTransparency = 0}):Play()
			TweenService:Create(Dropdown.UIStroke, TweenInfo.new(0.7, Enum.EasingStyle.Exponential), {Transparency = 0}):Play()
			TweenService:Create(Dropdown.Title, TweenInfo.new(0.7, Enum.EasingStyle.Exponential), {TextTransparency = 0}):Play()
			
			Dropdown.Interact.MouseButton1Click:Connect(function()
				if Debounce then return end
				local IsOpen = Dropdown.List.Visible
				if IsOpen then
					TweenService:Create(Dropdown, TweenInfo.new(0.5, Enum.EasingStyle.Exponential), {Size = UDim2.new(1, -10, 0, 45)}):Play()
					Dropdown.List.Visible = false
					TweenService:Create(Dropdown.Toggle, TweenInfo.new(0.7, Enum.EasingStyle.Exponential), {Rotation = 180}):Play()
				else
					TweenService:Create(Dropdown, TweenInfo.new(0.5, Enum.EasingStyle.Exponential), {Size = UDim2.new(1, -10, 0, 180)}):Play()
					Dropdown.List.Visible = true
					TweenService:Create(Dropdown.Toggle, TweenInfo.new(0.7, Enum.EasingStyle.Exponential), {Rotation = 0}):Play()
				end
			end)
			
			local Obj = {
				Set = function(Self, NewOpt: (string|{string}))
					if typeof(NewOpt) == "string" then NewOpt = {NewOpt} end
					DropdownSettings.CurrentOption = NewOpt
					UpdateDisplay()
					pcall(DropdownSettings.Callback, DropdownSettings.CurrentOption)
					if not DropdownSettings.Ext then SaveConfiguration() end
				end,
				Refresh = function(Self, NewOpts: {string})
					DropdownSettings.Options = NewOpts
					Populate()
				end
			}
			
			if DropdownSettings.Flag then
				DropdownSettings.Type = "Dropdown"
				RayfieldLibrary.Flags[DropdownSettings.Flag] = DropdownSettings
			end
			return Obj
		end
		
		function Tab:CreateKeybind(KeybindSettings: {Name: string, CurrentKeybind: string?, Flag: string?, HoldToInteract: boolean?, CallOnChange: boolean?, Callback: (string|boolean) -> (), Ext: boolean?})
			local Keybind = Elements.Template.Keybind:Clone()
			Keybind.Name = KeybindSettings.Name
			Keybind.Title.Text = KeybindSettings.Name
			Keybind.Parent = TabPage
			Keybind.Visible = true
			KeybindSettings.CurrentKeybind = KeybindSettings.CurrentKeybind or ""
			Keybind.KeybindFrame.KeybindBox.Text = KeybindSettings.CurrentKeybind
			
			TweenService:Create(Keybind, TweenInfo.new(0.7, Enum.EasingStyle.Exponential), {BackgroundTransparency = 0}):Play()
			TweenService:Create(Keybind.UIStroke, TweenInfo.new(0.7, Enum.EasingStyle.Exponential), {Transparency = 0}):Play()
			TweenService:Create(Keybind.Title, TweenInfo.new(0.7, Enum.EasingStyle.Exponential), {TextTransparency = 0}):Play()
			
			local Checking = false
			Keybind.KeybindFrame.KeybindBox.Focused:Connect(function() Checking = true; Keybind.KeybindFrame.KeybindBox.Text = "" end)
			Keybind.KeybindFrame.KeybindBox.FocusLost:Connect(function()
				Checking = false
				if Keybind.KeybindFrame.KeybindBox.Text == "" then Keybind.KeybindFrame.KeybindBox.Text = KeybindSettings.CurrentKeybind end
				if not KeybindSettings.Ext then SaveConfiguration() end
			end)
			
			local Conn = UserInputService.InputBegan:Connect(function(Inp, Gpe)
				if Checking and Inp.KeyCode ~= Enum.KeyCode.Unknown then
					local KeyName = Inp.KeyCode.Name
					Keybind.KeybindFrame.KeybindBox.Text = KeyName
					KeybindSettings.CurrentKeybind = KeyName
					Keybind.KeybindFrame.KeybindBox:ReleaseFocus()
					if KeybindSettings.CallOnChange then pcall(KeybindSettings.Callback, KeyName) end
				elseif not Checking and not Gpe and KeybindSettings.CurrentKeybind and Inp.KeyCode == Enum.KeyCode[KeybindSettings.CurrentKeybind] then
					if KeybindSettings.HoldToInteract then
						local Held = true
						local C
						C = Inp.Changed:Connect(function(Prop) if Prop == "UserInputState" then C:Disconnect(); Held = false end end)
						while Held do pcall(KeybindSettings.Callback, true); task.wait() end
						pcall(KeybindSettings.Callback, false)
					else
						pcall(KeybindSettings.Callback)
					end
				end
			end)
			table.insert(KeybindConnections, Conn)
			
			local Obj = {
				Set = function(Self, NewKey: string)
					KeybindSettings.CurrentKeybind = NewKey
					Keybind.KeybindFrame.KeybindBox.Text = NewKey
					if KeybindSettings.CallOnChange then pcall(KeybindSettings.Callback, NewKey) end
					if not KeybindSettings.Ext then SaveConfiguration() end
				end
			}
			
			if KeybindSettings.Flag then
				KeybindSettings.Type = "Keybind"
				RayfieldLibrary.Flags[KeybindSettings.Flag] = KeybindSettings
			end
			return Obj
		end

		function Tab:CreateColorPicker(ColorPickerSettings: {Name: string, Color: Color3, Flag: string?, Callback: (Color3) -> (), Ext: boolean?})
			local CP = Elements.Template.ColorPicker:Clone()
			CP.Name = ColorPickerSettings.Name
			CP.Title.Text = ColorPickerSettings.Name
			CP.Parent = TabPage
			CP.Visible = true
			
			local H, S, V = ColorPickerSettings.Color:ToHSV()
			local Opened = false
			local MainDrag, SliderDrag = false, false
			local Mouse = Players.LocalPlayer:GetMouse()
			
			local function UpdateDisplay()
				local Col = Color3.fromHSV(H, S, V)
				CP.CPBackground.Display.BackgroundColor3 = Col
				CP.CPBackground.MainCP.MainPoint.Position = UDim2.new(S, -CP.CPBackground.MainCP.MainPoint.AbsoluteSize.X/2, 1-V, -CP.CPBackground.MainCP.MainPoint.AbsoluteSize.Y/2)
				CP.CPBackground.MainCP.MainPoint.ImageColor3 = Col
				CP.CPBackground.BackgroundColor3 = Color3.fromHSV(H, 1, 1)
				CP.ColorSlider.SliderPoint.Position = UDim2.new(0, H * CP.ColorSlider.AbsoluteSize.X - CP.ColorSlider.SliderPoint.AbsoluteSize.X/2, 0.5, 0)
				CP.ColorSlider.SliderPoint.ImageColor3 = Color3.fromHSV(H, 1, 1)
				
				local R, G, B = math.floor(Col.R*255+0.5), math.floor(Col.G*255+0.5), math.floor(Col.B*255+0.5)
				CP.RGB.RInput.InputBox.Text = tostring(R)
				CP.RGB.GInput.InputBox.Text = tostring(G)
				CP.RGB.BInput.InputBox.Text = tostring(B)
				CP.HexInput.InputBox.Text = string.format("#%02X%02X%02X", R, G, B)
				
				ColorPickerSettings.Color = Col
			end
			
			UpdateDisplay()
			
			CP.Interact.MouseButton1Down:Connect(function()
				Opened = not Opened
				if Opened then
					TweenService:Create(CP, TweenInfo.new(0.6, Enum.EasingStyle.Exponential), {Size = UDim2.new(1, -10, 0, 120)}):Play()
					TweenService:Create(CP.CPBackground, TweenInfo.new(0.6, Enum.EasingStyle.Exponential), {Size = UDim2.new(0, 173, 0, 86), BackgroundTransparency = 0}):Play()
					TweenService:Create(CP.CPBackground.MainCP.MainPoint, TweenInfo.new(0.2, Enum.EasingStyle.Exponential), {ImageTransparency = 0}):Play()
				else
					TweenService:Create(CP, TweenInfo.new(0.6, Enum.EasingStyle.Exponential), {Size = UDim2.new(1, -10, 0, 45)}):Play()
					TweenService:Create(CP.CPBackground, TweenInfo.new(0.6, Enum.EasingStyle.Exponential), {Size = UDim2.new(0, 39, 0, 22), BackgroundTransparency = 1}):Play()
					TweenService:Create(CP.CPBackground.MainCP.MainPoint, TweenInfo.new(0.2, Enum.EasingStyle.Exponential), {ImageTransparency = 1}):Play()
				end
			end)
			
			CP.CPBackground.MainCP.InputBegan:Connect(function(Inp) if Inp.UserInputType == Enum.UserInputType.MouseButton1 and Opened then MainDrag = true end end)
			CP.ColorSlider.InputBegan:Connect(function(Inp) if Inp.UserInputType == Enum.UserInputType.MouseButton1 then SliderDrag = true end end)
			UserInputService.InputEnded:Connect(function(Inp) if Inp.UserInputType == Enum.UserInputType.MouseButton1 then MainDrag = false; SliderDrag = false end end)
			
			local RenderConn = RunService.RenderStepped:Connect(function()
				if MainDrag then
					local Lx = math.clamp(Mouse.X - CP.CPBackground.MainCP.AbsolutePosition.X, 0, CP.CPBackground.MainCP.AbsoluteSize.X)
					local Ly = math.clamp(Mouse.Y - CP.CPBackground.MainCP.AbsolutePosition.Y, 0, CP.CPBackground.MainCP.AbsoluteSize.Y)
					S = Lx / CP.CPBackground.MainCP.AbsoluteSize.X
					V = 1 - (Ly / CP.CPBackground.MainCP.AbsoluteSize.Y)
					UpdateDisplay()
					pcall(ColorPickerSettings.Callback, ColorPickerSettings.Color)
					if not ColorPickerSettings.Ext then SaveConfiguration() end
				elseif SliderDrag then
					local Lx = math.clamp(Mouse.X - CP.ColorSlider.AbsolutePosition.X, 0, CP.ColorSlider.AbsoluteSize.X)
					H = Lx / CP.ColorSlider.AbsoluteSize.X
					UpdateDisplay()
					pcall(ColorPickerSettings.Callback, ColorPickerSettings.Color)
					if not ColorPickerSettings.Ext then SaveConfiguration() end
				end
			end)
			
			CP.Destroying:Connect(function() RenderConn:Disconnect() end)
			
			local Obj = {
				Set = function(Self, Col: Color3)
					H, S, V = Col:ToHSV()
					UpdateDisplay()
				end
			}
			
			if ColorPickerSettings.Flag then
				ColorPickerSettings.Type = "ColorPicker"
				RayfieldLibrary.Flags[ColorPickerSettings.Flag] = ColorPickerSettings
			end
			return Obj
		end

				local Sh = Elements.Template.SectionTitle:Clone()
				Sh.Name = "SearchHeader"
				Sh.Title.Text = "Results from '" .. Page.Name .. "'"
				Sh.LayoutOrder = -100
				Sh.Parent = Page
				Sh.Visible = true
			end
		else
			local Sh = Page:FindFirstChild("SearchHeader")
			if Sh then Sh:Destroy() end
		end
		
		for _, El in ipairs(Page:GetChildren()) do
			if El:IsA("GuiObject") and El.Name ~= "UIListLayout" and El.Name ~= "Placeholder" and El.Name ~= "SearchHeader" then
				if El.Name == "SectionTitle" then
					El.Visible = (#Query == 0)
				else
					El.Visible = El.Name:lower():find(Query, 1, true) ~= nil
				end
			end
		end
	end)
	
	Main.Search.Input.FocusLost:Connect(function()
		if #Main.Search.Input.Text == 0 and SearchOpen then task.wait(0.12); CloseSearch() end
	end)
	
	local HideConn = UserInputService.InputBegan:Connect(function(Inp, Gpe)
		if Gpe then return end
		local Key = GetSetting("General", "RayfieldOpen")
		if Key and Inp.KeyCode == Enum.KeyCode[Key] then
			if Hidden then Unhide() else Hide() end
		end
	end)
	
	if MPrompt then
		MPrompt.Interact.MouseButton1Click:Connect(function() if Hidden then Unhide() end end)
	end
	
	task.delay(4, function()
		if CEnabled then
			local Loaded = LoadConfiguration()
			if Loaded then RayfieldLibrary:Notify({Title = "Config Loaded", Content = "Previous session configuration restored."}) end
		end
		GlobalLoaded = true
	end)
	
	return Window
end

function RayfieldLibrary:SetVisibility(Vis: boolean)
	if Vis then Unhide() else Hide(false) end
end

function RayfieldLibrary:IsVisible(): boolean
	return not Hidden
end

function RayfieldLibrary:Destroy()
	RayfieldDestroyed = true
	for _, C in ipairs(KeybindConnections) do C:Disconnect() end
	Rayfield:Destroy()
end

LoadSettings()

return RayfieldLibrary
