local function ApplyHoverTween(Button, Color, Info)
    TweenService:Create(Button, Info, {BackgroundColor3 = Color}):Play()
end

local function AddHoverEffect(Button, EnterColor, LeaveColor)
    Button.MouseEnter:Connect(function() ApplyHoverTween(Button, EnterColor, HoverTweenEnter) end)
    Button.MouseLeave:Connect(function() ApplyHoverTween(Button, LeaveColor, HoverTweenLeave) end)
end

local function MakeDraggable(Trigger, Target)
    local Dragging = false
    local DragInput = nil
    local StartPos = Vector2.zero
    local StartTargetPos = UDim2.new()
    
    Trigger.InputBegan:Connect(function(Input)
        if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then
            Dragging = true
            StartPos = Input.Position
            StartTargetPos = Target.Position
            Input.Changed:Connect(function()
                if Input.UserInputState == Enum.UserInputState.End then 
                    Dragging = false 
                end
            end)
        end
    end)
    
    Trigger.InputChanged:Connect(function(Input)
        if Input.UserInputType == Enum.UserInputType.MouseMovement or Input.UserInputType == Enum.UserInputType.Touch then
            DragInput = Input
        end
    end)
    
    UserInputService.InputChanged:Connect(function(Input)
        if Dragging and Input == DragInput then
            local Delta = Input.Position - StartPos
            Target.Position = UDim2.new(StartTargetPos.X.Scale, StartTargetPos.X.Offset + Delta.X, StartTargetPos.Y.Scale, StartTargetPos.Y.Offset + Delta.Y)
        end
    end)
end
