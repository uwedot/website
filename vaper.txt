local function MakeDraggable(topbar, object)
	local Dragging, DragInput, DragStart, StartPosition
	topbar.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			Dragging = true
			DragStart = input.Position
			StartPosition = object.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then Dragging = false end
			end)
		end
	end)
	topbar.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			DragInput = input
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if input == DragInput and Dragging then
			local Delta = input.Position - DragStart
			object.Position = UDim2.new(StartPosition.X.Scale, StartPosition.X.Offset + Delta.X, StartPosition.Y.Scale, StartPosition.Y.Offset + Delta.Y)
		end
	end)
end

			local binding = false
			local Key = keypreset.Name
			local B = Instance.new("TextButton")
			B.Parent = Tab
			B.BackgroundColor3 = Color3.fromRGB(34, 34, 34)
			B.Size = UDim2.new(0, 363, 0, 42)
			B.AutoButtonColor = false
			Instance.new("UICorner", B).CornerRadius = UDim.new(0, 5)
			local BT = Instance.new("TextLabel")
			BT.Parent = B
			BT.BackgroundTransparency = 1
			BT.Position = UDim2.new(0.0358, 0, 0, 0)
			BT.Size = UDim2.new(0, 187, 0, 42)
			BT.Font = Enum.Font.Gotham
			BT.Text = text
			BT.TextColor3 = Color3.new(1,1,1)
			BT.TextSize = 14
			BT.TextXAlignment = Enum.TextXAlignment.Left
			local KText = Instance.new("TextLabel")
			KText.Parent = B
			KText.BackgroundTransparency = 1
			KText.Position = UDim2.new(0.0358, 0, 0, 0)
			KText.Size = UDim2.new(0, 337, 0, 42)
			KText.Font = Enum.Font.Gotham
			KText.Text = Key
			KText.TextColor3 = Color3.new(1,1,1)
			KText.TextSize = 14
			KText.TextXAlignment = Enum.TextXAlignment.Right

			B.MouseButton1Click:Connect(function()
				KText.Text = "..."
				binding = true
				local i = UserInputService.InputBegan:Wait()
				if i.KeyCode.Name ~= "Unknown" then
					Key = i.KeyCode.Name
					KText.Text = Key
				end
				binding = false
			end)

			UserInputService.InputBegan:Connect(function(i, gpe)
				if not gpe and not binding and i.KeyCode.Name == Key then pcall(callback) end
			end)
			UpdateCanvas()
		end
